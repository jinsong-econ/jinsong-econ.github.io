    // 访客计数器核心逻辑（本地持久化 + 自动对接不蒜子全网统计）

    function initVisitorCounter() {

      const storageKeyPV = 'songjin_site_local_pv';

      const storageKeyUV = 'songjin_site_local_uv';

      const storageKeyMyVisits = 'songjin_my_visit_count';

 

      let localPV = parseInt(localStorage.getItem(storageKeyPV) || '0', 10);

      let localUV = parseInt(localStorage.getItem(storageKeyUV) || '0', 10);

      let myVisits = parseInt(localStorage.getItem(storageKeyMyVisits) || '0', 10);

 

      // 每次打开或刷新页面，浏览量 +1

      myVisits += 1;

      localPV += 1;

      if (myVisits === 1) {

        localUV += 1; // 首次访问计入独立访客

      }

 

      localStorage.setItem(storageKeyPV, localPV);

      localStorage.setItem(storageKeyUV, localUV);

      localStorage.setItem(storageKeyMyVisits, myVisits);

 

      const displayPV = PROFILE_DATA.visitorConfig.basePV + localPV;

      const displayUV = PROFILE_DATA.visitorConfig.baseUV + localUV;

 

      updateVisitorDOM(displayPV, displayUV, myVisits);

 

      // 监听不蒜子（Busuanzi）全网真实统计返回结果（部署到 GitHub Pages 后自动触发）

      let checkCount = 0;

      const timer = setInterval(() => {

        checkCount++;

        const bszPVEl = document.getElementById('busuanzi_value_site_pv');

        const bszUVEl = document.getElementById('busuanzi_value_site_uv');

        if (bszPVEl && bszPVEl.textContent && parseInt(bszPVEl.textContent, 10) > 0) {

          const realPV = parseInt(bszPVEl.textContent, 10);

          const realUV = parseInt(bszUVEl.textContent || '1', 10);

          // 取全网真实数据与本地基准中的较大值，确保刚上线时数字也优雅好看

          updateVisitorDOM(

            Math.max(realPV, displayPV),

            Math.max(realUV, displayUV),

            myVisits

          );

          clearInterval(timer);

        }

        if (checkCount > 12) clearInterval(timer);

      }, 500);

    }

 

    function updateVisitorDOM(pv, uv, myVisits) {

      const formattedPV = pv.toLocaleString();

      const formattedUV = uv.toLocaleString();

 

      const sidePV = document.getElementById('sidebar-pv-num');

      const sideUV = document.getElementById('sidebar-uv-num');

      const mainPV = document.getElementById('main-pv-num');

      const mainUV = document.getElementById('main-uv-num');

      const localEl = document.getElementById('local-visit-num');

      const footerPV = document.getElementById('footer-pv-num');

 

      if (sidePV) sidePV.textContent = formattedPV;

      if (sideUV) sideUV.textContent = formattedUV;

      if (mainPV) mainPV.textContent = formattedPV;

      if (mainUV) mainUV.textContent = formattedUV;

      if (localEl) localEl.textContent = myVisits.toLocaleString();

      if (footerPV) footerPV.textContent = formattedPV;

    }
